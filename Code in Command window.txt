%% =========================================================
%  Final report figures for Chapter 4
%  Corresponding to:
%  Figure 4.1 Seasonal mean grid import power profiles
%  Figure 4.2 Seasonal mean grid export power profiles
%  Figure 4.3 Seasonal mean battery charging power profiles
%  Figure 4.4 Seasonal mean battery discharging power profiles
%  Figure 4.5 Seasonal mean battery state-of-charge profiles
%% =========================================================

fig_folder = 'final_report_figures';

if ~exist(fig_folder, 'dir')
    mkdir(fig_folder);
end

strategy_names = {'Day-ahead MILP', 'MPC', 'Rule-based'};
season_titles  = {'Spring', 'Summer', 'Autumn', 'Winter'};

%% =========================================================
%  Figure 4.1 Seasonal mean grid import power profiles
%% =========================================================

figure('Name', 'Figure 4.1 Seasonal mean grid import power profiles', ...
       'Position', [100 100 1200 800]);

tiledlayout(2, 2, 'TileSpacing', 'compact', 'Padding', 'compact');

for s = 1:numel(season_names)

    SEASON_now = season_names{s};
    season_idx = valid_all & (Season_label' == string(SEASON_now));

    if ~any(season_idx)
        continue;
    end

    Pbuy_da_mean_season   = mean(Pbuy_dayahead_all(:, season_idx), 2, 'omitnan');
    Pbuy_mpc_mean_season  = mean(Pbuy_mpc_all(:, season_idx), 2, 'omitnan');
    Pbuy_rule_mean_season = mean(Pbuy_rule_all(:, season_idx), 2, 'omitnan');

    nexttile;
    plot(1:T, Pbuy_da_mean_season, '-o', 'LineWidth', 1.2); hold on;
    plot(1:T, Pbuy_mpc_mean_season, '-s', 'LineWidth', 1.2);
    plot(1:T, Pbuy_rule_mean_season, '-^', 'LineWidth', 1.2);

    title(season_titles{s});
    xlabel('Time of day');
    ylabel('Grid import power (kW)');

    xticks(1:tick_step:T);
    xticklabels(time_lbl_48(1:tick_step:T));
    xtickangle(45);

    grid on;

    if s == 1
        legend(strategy_names, 'Location', 'best');
    end

end

sgtitle('Seasonal mean grid import power profiles for the three EMS strategies');

exportgraphics(gcf, fullfile(fig_folder, 'Figure_4_1_seasonal_mean_grid_import.png'), ...
    'Resolution', 300);
